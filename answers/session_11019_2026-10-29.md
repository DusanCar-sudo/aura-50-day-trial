def http_get(url, max_retries=3):
    """
    Perform an HTTP GET request on `url`, retrying up to `max_retries` times.
    
    Args:
        url: str — the URL to request.
        max_retries: int — number of retry attempts (default 3).
    
    Returns:
        The response body as a string, or None if all retries fail.
    """
    delay = 1  # starting backoff in seconds
    for attempt in range(max_retries):
        try:
            with urllib.request.urlopen(url) as response:
                return response.read().decode('utf-8')
        except Exception as e:
            if attempt < max_retries - 1:
                time.sleep(delay)
                delay *= 2  # exponential backoff
            else:
                print(f"Failed after {max_retries} attempts: {e}")
    return None

import { createReadStream } from "node:fs";
import { createInterface } from "node:readline";

/**
 * Stream a file line by line. Memory stays flat regardless of file size:
 * the stream reads in fixed-size chunks and readline emits one line at a time.
 * Returns the number of lines seen.
 */
export async function readLines(
  path: string,
  onLine: (line: string, lineNo: number) => void | Promise<void>,
): Promise<number> {
  const stream = createReadStream(path, { encoding: "utf8" });
  const rl = createInterface({ input: stream, crlfDelay: Infinity });

  let lineNo = 0;
  try {
    for await (const line of rl) {
      lineNo++;
      await onLine(line, lineNo);
    }
  } finally {
    rl.close();
    stream.destroy();
  }
  return lineNo;
}

"""Stream a file line by line without loading it into memory.

Iterating a file object yields one line at a time; Python reads in buffered
chunks under the hood, so memory stays flat for arbitrarily large files.
"""

from typing import Callable, Iterator


def read_lines(path: str) -> Iterator[str]:
    """Yield lines from `path` one at a time, newline stripped."""
    with open(path, "r", encoding="utf-8", errors="replace") as fh:
        for line in fh:
            yield line.rstrip("\n")


def for_each_line(path: str, on_line: Callable[[str, int], None]) -> int:
    """Call `on_line(line, line_no)` for every line. Returns the line count."""
    count = 0
    for count, line in enumerate(read_lines(path), start=1):
        on_line(line, count)
    return count

package main

import (
	"bufio"
	"fmt"
	"io"
	"os"
)

// ReadLines streams a file line by line. bufio.Scanner holds only one line
// (plus its buffer) in memory at a time, so file size does not matter.
// Returns the number of lines read.
func ReadLines(path string, onLine func(line string, lineNo int) error) (int, error) {
	f, err := os.Open(path)
	if err != nil {
		return 0, err
	}
	defer f.Close()

	sc := bufio.NewScanner(f)
	// Default max token size is 64KB; raise it for long lines.
	sc.Buffer(make([]byte, 0, 64*1024), 16*1024*1024)

	lineNo := 0
	for sc.Scan() {
		lineNo++
		if err := onLine(sc.Text(), lineNo); err != nil {
			return lineNo, err
		}
	}
	if err := sc.Err(); err != nil && err != io.EOF {
		return lineNo, err
	}
	return lineNo, nil
}

func main() {
	if len(os.Args) < 2 {
		fmt.Fprintln(os.Stderr, "usage: readlines <file>")
		os.Exit(1)

		if shown < 3 {

		}
		return nil
	})
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}
	fmt.Printf("total lines: %d\n", total)
}

def iter_lines(path: str, encoding: str = "utf-8") -> Iterator[str]:

	"os"

// ReadLines streams path line by line. The scanner holds one line at a time,
// so memory use is bounded by the longest line, not the file size.
func ReadLines(path string, onLine func(line string, lineNumber int) error) (int, error) {
	f, err := os.Open(path)
	if err != nil {
		return 0, err
	}
	defer f.Close()

	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 0, 64*1024), 16*1024*1024) // allow long lines

	count := 0
	for sc.Scan() {
		count++
		if err := onLine(sc.Text(), count); err != nil {
			return count, err
		}
	}
	if err := sc.Err(); err != nil {
		return count, fmt.Errorf("scan %s: %w", path, err)
	}
	return count, nil
}

func main() {
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: readlines <file>")

 * - `undefined` in `override` does NOT erase a base value (use `null` to clear).
 * - Neither input is mutated; the result is a fresh object graph.
 */
export type PlainObject = Record<string, unknown>;

export function isPlainObject(value: unknown): value is PlainObject {
  if (value === null || typeof value !== 'object') return false;
  const proto = Object.getPrototypeOf(value);
  return proto === Object.prototype || proto === null;
}

export function deepMerge<T extends PlainObject, U extends PlainObject>(
  base: T,
  override: U,
): T & U {
  const out: PlainObject = { ...base };

  for (const [key, value] of Object.entries(override)) {
    if (value === undefined) continue;

    const current = out[key];
    out[key] = isPlainObject(current) && isPlainObject(value)
      ? deepMerge(current, value)
      : value;
  }

  return out as T & U;
}

export function flatten<T>(input: readonly unknown[], depth: number = Infinity): T[] {

// It is iterative (explicit stack) so arbitrarily deep input cannot overflow

// slices and other slice types are treated as leaf values.

		return append(out, input...)
	}

		arr []any
		i   int
		d   int

/// Flatten a deeply nested `Vec<Value>` into a single level.
///
/// Iterative (explicit stack) so arbitrarily deep input cannot overflow the
/// thread stack. Only `Value::Array` is descended into; every other variant is
/// a leaf.
///

/// The input is consumed, so no clone is needed.

pub enum Value {
    Null,
    Bool(bool),
    Int(i64),
    Float(f64),
    Str(String),
    Array(Vec<Value>),
}

pub fn flatten(input: Vec<Value>, depth: Option<usize>) -> Vec<Value> {
    let mut out = Vec::with_capacity(input.len());
    let max_depth = depth.unwrap_or(usize::MAX);
    if max_depth == 0 {
        return input;
    }

    // Each frame is (array, index, remaining_depth).
    let mut stack: Vec<(Vec<Value>, usize, usize)> = vec![(input, 0, max_depth)];

    while let Some(top) = stack.last_mut() {
        let (arr, i, d) = top;

        if *i >= arr.len() {
            stack.pop();
            continue;

        // Take the element out by swapping with Null so we own it without cloning.

        *i += 1;
        let remaining = *d;

        match value {
            Value::Array(inner) if remaining > 1 => {
                stack.push((inner, 0, remaining - 1));
            }
            other => out.push(other),
        }
    }

    out
}

/**
 * Concurrent worker pool: N jobs, fixed number of workers.
 *
 * Design notes
 * - Workers pull from a shared index counter (no per-job promise allocation),
 *   so a 1M-job run costs the same memory as a 10-job run.
 * - `limit` caps in-flight jobs; `stopOnError` cancels the remaining queue.
 * - Results come back in job order regardless of completion order.
 */

export interface PoolOptions {
  /** Max jobs running at once. Defaults to 4. */
  concurrency?: number;
  /** Abort remaining jobs on the first rejection. Defaults to false. */
  stopOnError?: boolean;
  /** Called after each job settles (success or failure). */
  onSettled?: (index: number, error?: unknown) => void;
}

export interface PoolResult<R> {
  results: R[];
  errors: { index: number; error: unknown }[];
}

export async function runPool<T, R>(
  jobs: readonly T[],
  worker: (job: T, index: number) => Promise<R> | R,
  options: PoolOptions = {},
): Promise<PoolResult<R>> {
  const concurrency = Math.max(1, Math.floor(options.concurrency ?? 4));
  const stopOnError = options.stopOnError ?? false;

  const results = new Array<R>(jobs.length);
  const errors: { index: number; error: unknown }[] = [];
  let next = 0;
  let aborted = false;

  const runOne = async (): Promise<void> => {
    while (!aborted) {
      const index = next++;
      if (index >= jobs.length) return;
      try {
        results[index] = await worker(jobs[index], index);
        options.onSettled?.(index);
      } catch (error) {
        errors.push({ index, error });
        options.onSettled?.(index, error);
        if (stopOnError) {
          aborted = true;
          return;
        }
      }
    }
  };

  const workers = Array.from(
    { length: Math.min(concurrency, jobs.length) },
    () => runOne(),
  );
  await Promise.all(workers);

  return { results, errors };
}

//
//	stage A (produce) -> stage B (transform) -> stage C (consume)
//
// Every stage runs in its own goroutine and communicates only through
// channels. Cancellation and error propagation flow through a shared context,
// so a failure in any stage tears the whole pipeline down without leaking
// goroutines.
package main

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/signal"
	"strings"
	"sync"
	"syscall"
	"time"

// Item is the unit of work that travels through the pipeline.
type Item struct {
	ID    int
	Value string
}

// Result is what stage C emits after consuming a transformed item.
type Result struct {

	Value string
	Len   int

// produce is stage A: it generates items and closes the output channel when
// done. It respects ctx cancellation so a downstream failure stops it early.
func produce(ctx context.Context, n int) <-chan Item {
	out := make(chan Item)
	go func() {
		defer close(out)
		for i := 1; i <= n; i++ {

			select {
			case out <- item:
			case <-ctx.Done():
				return
			}

			case <-time.After(20 * time.Millisecond):

	}()

			if !ok {

			}

/**
 * Longest Common Subsequence (LCS).
 *
 * Returns one longest common subsequence of `a` and `b` (a subsequence keeps
 * order but need not be contiguous). Ties are broken deterministically:
 * when skipping a character from `a` and from `b` are equally good, we skip
 * from `a` first.
 *
 * Time:  O(n*m)   Space: O(n*m) for the DP table (O(min(n,m)) if only the
 * length is needed — see lcsLength).
 */
export function lcs(a: string, b: string): string {
  const n = a.length;
  const m = b.length;

  // dp[i][j] = LCS length of a[i:] and b[j:]
  const dp: number[][] = Array.from({ length: n + 1 }, () =>
    new Array<number>(m + 1).fill(0),
  );

  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      dp[i][j] =
        a[i] === b[j]
          ? dp[i + 1][j + 1] + 1
          : Math.max(dp[i + 1][j], dp[i][j + 1]);
    }
  }

  // Walk the table forward, collecting matched characters.
  let out = "";
  let i = 0;
  let j = 0;
  while (i < n && j < m) {
    if (a[i] === b[j]) {
      out += a[i];
      i++;
      j++;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) {
      i++;
    } else {
      j++;
    }
  }
  return out;
}

/** LCS length only — O(min(n, m)) extra space. */
export function lcsLength(a: string, b: string): number {
  if (b.length > a.length) [a, b] = [b, a]; // b is the shorter one
  const m = b.length;
  let prev = new Array<number>(m + 1).fill(0);
  let cur = new Array<number>(m + 1).fill(0);

  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= m; j++) {
      cur[j] =
        a[i - 1] === b[j - 1]
          ? prev[j - 1] + 1
          : Math.max(prev[j], cur[j - 1]);
    }
    [prev, cur] = [cur, prev];
  }
  return prev[m];
}

"""Longest Common Subsequence (LCS).

Returns one longest common subsequence of `a` and `b` (a subsequence keeps
order but need not be contiguous). Ties are broken deterministically: when
skipping a character from `a` and from `b` are equally good, we skip from `a`
first — the same rule as the TypeScript and Go versions.

Time:  O(n*m)   Space: O(n*m) for the DP table (O(min(n,m)) if only the
length is needed — see lcs_length).
"""

from __future__ import annotations


def lcs(a: str, b: str) -> str:
    """Return one longest common subsequence of a and b."""
    n, m = len(a), len(b)

    # dp[i][j] = LCS length of a[i:] and b[j:]
    dp = [[0] * (m + 1) for _ in range(n + 1)]

    for i in range(n - 1, -1, -1):

// Longest Common Subsequence (LCS).
//
// Returns one longest common subsequence of a and b (a subsequence keeps order
// but need not be contiguous). Ties are broken deterministically: when skipping
// a character from a and from b are equally good, we skip from a first — the
// same rule as the TypeScript and Python versions.
//
// Time:  O(n*m)   Space: O(n*m) for the DP table (O(min(n,m)) if only the
// length is needed — see LCSLength).
package main

import "fmt"

// LCS returns one longest common subsequence of a and b.
func LCS(a, b string) string {
	ar, br := []rune(a), []rune(b) // runes, so multi-byte input is handled
	n, m := len(ar), len(br)

	// dp[i][j] = LCS length of ar[i:] and br[j:]
	dp := make([][]int, n+1)
	for i := range dp {
		dp[i] = make([]int, m+1)
	}

	for i := n - 1; i >= 0; i-- {
		for j := m - 1; j >= 0; j-- {
			if ar[i] == br[j] {
				dp[i][j] = dp[i+1][j+1] + 1
			} else if dp[i+1][j] >= dp[i][j+1] {
				dp[i][j] = dp[i+1][j]
			} else {
				dp[i][j] = dp[i][j+1]
			}
		}
	}

	// Walk the table forward, collecting matched characters.
	out := make([]rune, 0, dp[0][0])
	i, j := 0, 0
	for i < n && j < m {
		switch {
		case ar[i] == br[j]:
			out = append(out, ar[i])
			i++
			j++
		case dp[i+1][j] >= dp[i][j+1]:
			i++
		default:
			j++
		}
	}
	return string(out)
}

				cur[j] = prev[j-1] + 1

			} else {

			}
		}

	}
	return prev[m]
}

 * Floyd's cycle detection ("tortoise and hare") for singly linked lists.
 *
 * Time:  O(n)  — the hare advances at most n steps before either reaching the
 *                 end or meeting the tortoise inside the cycle.
 * Space: O(1)  — two pointers, no visited set.
 *
 * The second phase (finding the cycle entry) is the part people forget:
 * after the meeting point, reset one pointer to the head and advance both one
 * step at a time; they meet exactly at the first node of the cycle.
 */

export class ListNode<T> {
  value: T;
  next: ListNode<T> | null = null;

  constructor(value: T) {
    this.value = value;
  }
}

/** Build a list from values; if cycleAt >= 0, the tail links back to that index. */
export function buildList<T>(values: T[], cycleAt = -1): ListNode<T> | null {
  if (values.length === 0) return null;
  const nodes = values.map((v) => new ListNode(v));
  for (let i = 0; i < nodes.length - 1; i++) nodes[i].next = nodes[i + 1];
  if (cycleAt >= 0 && cycleAt < nodes.length) {
    nodes[nodes.length - 1].next = nodes[cycleAt];
  }
  return nodes[0];
}

/** True if the list contains a cycle. */
export function hasCycle<T>(head: ListNode<T> | null): boolean {
  let slow = head;
  let fast = head;

  while (fast !== null && fast.next !== null) {
    slow = slow!.next;
    fast = fast.next.next;
    if (slow === fast) return true;
  }
  return false;
}

/**
 * Returns the node where the cycle begins, or null if there is no cycle.
 * Phase 1 finds a meeting point; phase 2 finds the entry.
 */
export function findCycleStart<T>(head: ListNode<T> | null): ListNode<T> | null {
  let slow = head;
  let fast = head;

  // Phase 1: detect the meeting point.
  while (fast !== null && fast.next !== null) {
    slow = slow!.next;
    fast = fast.next.next;
    if (slow === fast) break;
  }
  if (fast === null || fast.next === null) return null; // no cycle

  // Phase 2: head and meeting point are both `k` steps from the entry,
  // where k = distance from head to the cycle entry.
  slow = head;
  while (slow !== fast) {
    slow = slow!.next;
    fast = fast!.next;
  }
  return slow;
}

  if (start === null) return 0;

  let length = 1;
  let node = start.next;
  while (node !== start) {
    node = node!.next;
    length++;
  }
  return length;

package floyd

// Node is a singly linked list node.
type Node[T any] struct {
	Value T
	Next  *Node[T]
}

// BuildList builds a list from values. If cycleAt >= 0, the tail links back to
// that index, creating a cycle.
func BuildList[T any](values []T, cycleAt int) *Node[T] {
	if len(values) == 0 {
		return nil
	}
	nodes := make([]*Node[T], len(values))
	for i, v := range values {
		nodes[i] = &Node[T]{Value: v}
	}
	for i := 0; i < len(nodes)-1; i++ {
		nodes[i].Next = nodes[i+1]
	}
	if cycleAt >= 0 && cycleAt < len(nodes) {
		nodes[len(nodes)-1].Next = nodes[cycleAt]
	}
	return nodes[0]
}

func HasCycle[T any](head *Node[T]) bool {
	slow, fast := head, head
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
		if slow == fast {
			return true
		}
	}
	return false
}

// no cycle.

	}

/**
 * Trie (prefix tree) with insert / search / startsWith.
 *
 * Complexity, where L = length of the key and A = alphabet size:
 *   insert      O(L)   — one node hop per character
 *   search      O(L)   — walk L nodes, then check the terminal flag
 *   startsWith  O(L)   — walk L nodes, no terminal check
 *   space       O(total characters stored) worst case, shared prefixes stored once
 *
 * Children are a Map, so each node costs O(1) per character regardless of
 * alphabet size. A fixed 26-slot array is faster but only works for a-z and
 * wastes memory on sparse nodes.
 */

class TrieNode {
  readonly children = new Map<string, TrieNode>();
  isEnd = false;
}

export class Trie {
  private readonly root = new TrieNode();
  private _size = 0;

  /** Number of distinct keys inserted. */
  get size(): number {
    return this._size;
  }

  /** Insert a key. Returns false if it was already present. */
  insert(word: string): boolean {
    let node = this.root;
    for (const ch of word) {
      let next = node.children.get(ch);
      if (next === undefined) {
        next = new TrieNode();
        node.children.set(ch, next);
      }
      node = next;
    }
    if (node.isEnd) return false;
    node.isEnd = true;
    this._size++;
    return true;
  }

  /** True only if the exact key was inserted (not merely a prefix of one). */
  search(word: string): boolean {
    const node = this.walk(word);
    return node !== undefined && node.isEnd;
  }

  /** True if any inserted key begins with the given prefix. */
  startsWith(prefix: string): boolean {
    return this.walk(prefix) !== undefined;
  }

  /** Walk to the node for `s`, or undefined if the path breaks. */
  private walk(s: string): TrieNode | undefined {
    let node: TrieNode | undefined = this.root;
    for (const ch of s) {
      node = node.children.get(ch);
      if (node === undefined) return undefined;
    }
    return node;
  }
}
